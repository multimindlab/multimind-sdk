# MultiMind SDK — CI / reproducible install requirements.
#
# Source of truth for dependencies is pyproject.toml. This file exists only
# so CI (and contributors who want a fully populated env) can run a single
# `pip install -r requirements.txt` to install the package together with all
# optional extras and dev tooling.
#
# For everyday usage prefer the extras directly, e.g.:
#   pip install multimind-sdk           # core only
#   pip install multimind-sdk[rag]      # core + RAG
#   pip install multimind-sdk[all]      # everything
#   pip install -e .[dev]               # editable dev install
#
# To regenerate a fully-pinned lockfile run:
#   pip install pip-tools
#   pip-compile --extra all --extra dev --output-file requirements.lock pyproject.toml

-e .[all,dev]
